{{/* Pages are generated by the _content.gotmpl content adapters and have no backing file, so Hugo's built-in relative .md link resolution can't see them. This hook resolves a link such as [Functions](functions.md#heading) by matching the link's basename against each generated page's sourceBasename param, so the same relative links keep working on GitHub and on the site. */}} {{- $dest := .Destination -}} {{- $parts := split $dest "#" -}} {{- $path := index $parts 0 -}} {{- $fragment := "" -}} {{- if gt (len $parts) 1 }}{{ $fragment = printf "#%s" (index $parts 1) }}{{ end -}} {{- $href := $dest -}} {{- if strings.HasSuffix $path ".md" -}} {{- $base := path.Base $path -}} {{- with (where site.RegularPages "Params.sourceBasename" $base) -}} {{- $href = printf "%s%s" (index . 0).RelPermalink $fragment -}} {{- end -}} {{- end -}} {{ .Text }}
