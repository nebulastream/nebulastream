include(${PROJECT_SOURCE_DIR}/cmake/PluginRegistrationUtil.cmake)

add_plugin_as_library(NEAREST_APPROACH_DISTANCE LogicalFunction nearest_approach_distance_logical_function NearestApproachDistanceLogicalFunction.cpp)
add_plugin_as_library(NEAREST_APPROACH_DISTANCE_TemporalPoint_TemporalPoint PhysicalFunction tpoint_nearest_approach_distance_physical_function TemporalPointNearestApproachDistancePhysicalFunction.cpp)
add_plugin_as_library(NEAREST_APPROACH_DISTANCE_TemporalPointSequence_TemporalPointSequence PhysicalFunction tpointsequence_nearest_approach_distance_physical_function TemporalPointSequenceNearestApproachDistancePhysicalFunction.cpp)
add_plugin_as_library(TGEO_AT_STBOX LogicalFunction tgeo_at_stbox_logical_function TgeoAtStboxLogicalFunction.cpp)
add_plugin_as_library(TGEO_AT_STBOX_TemporalPoint_SpatioTemporalBox PhysicalFunction tgeo_at_stbox_tpoint_physical_function TpointAtStboxPhysicalFunction.cpp)
add_plugin_as_library(TGEO_AT_STBOX_TemporalPointSequence_SpatioTemporalBox PhysicalFunction tgeo_at_stbox_tpointsequence_physical_function TpointSequenceAtStboxPhysicalFunction.cpp)
add_plugin_as_library(TEMPORAL_EINTERSECTS_GEOMETRY LogicalFunction temporal_eintersects_geometry_logical_function TemporalEIntersectsGeometryLogicalFunction.cpp)
add_plugin_as_library(TEMPORAL_EINTERSECTS_GEOMETRY_TemporalPoint_Polygon PhysicalFunction temporal_eintersects_geometry_tpoint_polygon_physical_function TPointEIntersectsPolygonPhysicalFunction.cpp)

target_include_directories(nearest_approach_distance_logical_function PRIVATE .)
target_include_directories(tpoint_nearest_approach_distance_physical_function PRIVATE .)
target_include_directories(tpointsequence_nearest_approach_distance_physical_function PRIVATE .)
target_include_directories(tgeo_at_stbox_logical_function PRIVATE .)
target_include_directories(tgeo_at_stbox_tpoint_physical_function PRIVATE .)
target_include_directories(tgeo_at_stbox_tpointsequence_physical_function PRIVATE .)
target_include_directories(temporal_eintersects_geometry_logical_function PRIVATE .)
target_include_directories(temporal_eintersects_geometry_tpoint_polygon_physical_function PRIVATE .)

target_link_libraries(tpointsequence_nearest_approach_distance_physical_function PUBLIC nes-meos)
target_link_libraries(tpoint_nearest_approach_distance_physical_function PUBLIC nes-meos)
target_link_libraries(tgeo_at_stbox_tpoint_physical_function PUBLIC nes-meos)
target_link_libraries(tgeo_at_stbox_tpointsequence_physical_function PUBLIC nes-meos)
target_link_libraries(temporal_eintersects_geometry_tpoint_polygon_physical_function PUBLIC nes-meos)

add_unreflection_plugin(LogicalFunction NearestApproachDistance KEY NEAREST_APPROACH_DISTANCE)
add_unreflection_plugin(LogicalFunction TgeoAtStbox KEY TGEO_AT_STBOX)
add_unreflection_plugin(LogicalFunction TemporalEIntersectsGeometry KEY TEMPORAL_EINTERSECTS_GEOMETRY)
